def utc():return datetime.now(timezone.utc).isoformat()
def sha(path):
 h=hashlib.sha256()
 with Path(path).open('rb') as f:
  for z in iter(lambda:f.read(1024*1024),b''):h.update(z)
 return h.hexdigest()

def parse_native(text):

def validate_config(cfg,native):

 if cfg.get('cme_boundary_csv'):tracked.append(Path(cfg['cme_boundary_csv']))
 if name in pin.get('movie_cases',set()):tracked.extend([pin['movie_basis']['matrix_csv'],pin['movie_basis']['coordinate_receipt']])
 # Metadata is immutable preparation data. Completion/scientific outcome uses
 # separate candidate_status.json, so the runtime hash receipt stays valid.
 atomic(out/'input_code_provenance.json',{str(p.relative_to(ROOT)):sha(p) for p in tracked})
 atomic(out/'candidate_status.json',dict(status='unrun',execution_complete=False,scientific_gates=None,missing_capabilities=missing))
 return dict(name=name,out=out,cfg=cfg,missing=missing,existing=False)

def completion(out,cfg):
 result=dict(execution_complete=False,endpoint_trace=False,statistics_readable=False)
 try:
  stats=json.loads((out/'statistics.json').read_text());result['statistics_readable']=True
  import csv
  with (out/'ct_trace.csv').open() as f:rows=list(csv.DictReader(f))
  if not rows:raise ValueError('empty ct_trace')
  days=[float(x['day']) for x in rows];result['endpoint_trace']=abs(days[-1]-float(cfg['end_day']))<=1e-10 and abs(days[0]-float(cfg['start_day']))<=1e-10 and all(b>a for a,b in zip(days,days[1:]))
  if not result['endpoint_trace']:raise ValueError('ct_trace endpoints/order disagree')
  if int(stats['steps'])<=0 or int(stats['stage_calls'])!=2*int(stats['steps']):raise ValueError('step/stage count mismatch')
  if int(cfg.get('ecliptic_slice_pixels',0))>0:result['movie_outputs']=movie_completion(out,cfg,days)
  result['execution_complete']=True
  scale=max(abs(float(x['B_code_scale'])) for x in rows)
  tol=1e-10*max(scale,1e-10)
  tests=dict(no_floor_activation=stats.get('floor_activations')==0,strict_floor_policy=stats.get('strict_floor_abort') is True,positive_density=float(stats['minimum_density_cm3'])>0,positive_pressure=float(stats['minimum_pressure_code'])>0,all_prescribed_inner_MHD_characteristics_incoming=float(stats['minimum_inner_fast_Mach_actual_ghost_trace'])>1,stage_target_B_tracking=float(stats['max_stage_inner_target_B_error_code'])<=tol,end_target_B_tracking=float(stats['max_inner_target_B_error_code'])<=tol)
  result.update(scientific_gates=tests,scientific_gates_passed=all(tests.values()),magnetic_tracking_absolute_tolerance_code=tol,statistics=stats,scope='These gates separate finite execution/positivity/inner-characteristic/CT-target validity from scientific convergence and observation validation. Work mismatch and thermal entropy require separate review.')
 except (OSError,ValueError,KeyError,TypeError,SyntaxError) as e:result['completion_error']=str(e)
 return result

def movie_completion(out,cfg,accepted_days):
 """Check output coverage/headers without materializing a plane or 3D state."""
 import ast,bisect,csv,struct
 def records(name):
  with (out/name).open() as f:return list(csv.DictReader(f))
 schedule=records('ecliptic_schedule.csv');planes=records('snapshots.csv');volume=records('volume_snapshots.csv')
 start=float(cfg['output_start_day']);cadence=float(cfg['snapshot_days']);end=float(cfg['end_day']);count=max(0,math.floor((end-start)/cadence+1e-9)+1)
 if count>4096 or len(schedule)!=count:raise ValueError('Movie requested-epoch coverage is incomplete or exceeds guard')
 unique={}
 for i,row in enumerate(schedule):
  requested=float(row['requested_day']);actual=float(row['actual_day']);pos=bisect.bisect_left(accepted_days,actual-1e-10)
  if abs(requested-(start+i*cadence))>1e-9 or actual+1e-10<requested or pos==len(accepted_days) or abs(accepted_days[pos]-actual)>1e-10:raise ValueError('Movie request/actual epoch is inconsistent with accepted trace')
  unique[row['file']]=actual
 if len(planes)!=len(unique) or [(x['file'],float(x['day'])) for x in planes]!=list(unique.items()):raise ValueError('Movie plane manifest disagrees with requested-to-actual schedule')
 if len(volume)!=1 or abs(float(volume[0]['day'])-end)>1e-10 or volume[0]['file']!='snapshot_00000.npy' or not(out/volume[0]['file']).is_file():raise ValueError('Movie final3D volume manifest is missing or inconsistent')
 pixels=int(cfg['ecliptic_slice_pixels'])
 for row in planes:
  if Path(row['file']).name!=row['file']:raise ValueError('Movie plane path must be a local filename')
  path=out/row['file']
  with path.open('rb') as f:
   if f.read(8)!=b'\x93NUMPY\x01\x00':raise ValueError('Movie plane requires expected NPYv1 header')
   raw_length=f.read(2)
   if len(raw_length)!=2:raise ValueError('Truncated movie NPY header')
   length=struct.unpack('<H',raw_length)[0];header=ast.literal_eval(f.read(length).decode('ascii'))
  if header.get('shape')!=(pixels,pixels) or header.get('descr')!='<f4' or header.get('fortran_order') is not False or path.stat().st_size!=10+length+pixels*pixels*4:raise ValueError('Movie plane shape/type/length mismatch')
 meta=json.loads((out/'ecliptic_slice_metadata.json').read_text());runtime=json.loads((out/'feature_runtime.json').read_text())
 basis=[float(x) for x in Path(cfg['ecliptic_to_solar_matrix_csv']).read_text().replace(',',' ').split()]
 if meta['pixels']!=pixels or meta['ecliptic_to_solver_matrix_row_major']!=basis or any(abs(a-b)>1e-12 for a,b in zip(meta['extent_au'],[-float(cfg['ecliptic_extent_au']),float(cfg['ecliptic_extent_au'])]*2)):raise ValueError('Movie output basis/extent differs from pinned config')
 if runtime.get('timesteps_trimmed_for_output') is not False or runtime.get('ecliptic_frames')!=len(planes):raise ValueError('Movie runtime output receipt is inconsistent')
 return dict(requested_epochs=count,actual_frames=len(planes),plane_shape=[pixels,pixels],basis_matches=True,actual_epochs_in_accepted_trace=True,timesteps_trimmed_for_output=False,final3D_preserved=True)

def limits(cpu,as_bytes):
 os.nice(15);resource.setrlimit(resource.RLIMIT_CPU,(cpu,cpu+1));resource.setrlimit(resource.RLIMIT_AS,(as_bytes,as_bytes));resource.setrlimit(resource.RLIMIT_CORE,(0,0))
def terminate(proc):
 if proc.poll() is not None:return
 try:os.killpg(proc.pid,signal.SIGTERM)
 except ProcessLookupError:return
 try:proc.wait(timeout=2)
 except subprocess.TimeoutExpired:
  try:os.killpg(proc.pid,signal.SIGKILL)
  except ProcessLookupError:pass
  proc.wait()
def time_usage(path):
 text=path.read_text() if path.exists() else ''
 result={}
 for key,label,convert in [('user_seconds','User time (seconds)',float),('system_seconds','System time (seconds)',float),('maxrss_kb','Maximum resident set size (kbytes)',int)]:
  match=re.search(r'^\s*'+re.escape(label)+r':\s*([0-9.]+)\s*$',text,re.MULTILINE)
  result[key]=convert(match.group(1)) if match else None
 result['gnu_time_resource_complete']=all(result[k] is not None for k in ['user_seconds','system_seconds','maxrss_kb'])
 return result
def process_rss(proc):
 try:
  p=psutil.Process(proc.pid);return sum(x.memory_info().rss for x in [p]+p.children(recursive=True) if x.is_running())
 except psutil.Error:return 0

def run_batch(prepared,pin,args):
 pending=[x for x in prepared if not x['existing']];active={};finished=[];peak_aggregate=0
 for x in pending:
  if x['missing']:raise ValueError(x['name']+': unsupported prospective capabilities '+','.join(x['missing']))
 progress=ROOT/'monitoring/ct_progress.json'
 try:
  while pending or active:
   while pending and len(active)<args.parallel:
    job=pending.pop(0);out=job['out'];
    # Archive any old marker immediately before launch; do not trust file mtimes
    # across Windows/WSL, and do not allow native exception-return0 to pass.
    if (out/'statistics.json').exists():raise ValueError('Unexpected preexisting completion marker; use --force')
    log=(out/'run.log').open('w');cmd=['/usr/bin/time','-v','-o',str(out/'resource_usage.txt'),str(pin['binary']),'-i',str(out/'athinput.ct')]
    started=time.monotonic();started_utc=utc()
    proc=subprocess.Popen(cmd,cwd=out,stdout=log,stderr=subprocess.STDOUT,start_new_session=True,preexec_fn=functools.partial(limits,args.cpu_limit,args.as_mib*2**20),env={**{k:v for k,v in os.environ.items() if not(k.startswith('OMP_') or k.startswith('GOMP_'))},'OMP_NUM_THREADS':str(job['cfg']['threads']),'OPENBLAS_NUM_THREADS':'1','MKL_NUM_THREADS':'1','OMP_DYNAMIC':'FALSE',**pin['runtime_env']})
    job.update(command=cmd,proc=proc,log=log,start=started,started_utc=started_utc,peak_rss=0,termination_reason=None);active[job['name']]=job
    atomic(out/'candidate_status.json',dict(status='running',execution_complete=False,started_utc=job['started_utc'],pid=proc.pid))
   aggregate=0
   for job in active.values():
    resident=process_rss(job['proc']);aggregate+=resident;job['peak_rss']=max(job['peak_rss'],resident)
    if resident>args.rss_mib*2**20:job['termination_reason']='per-job process-tree RSS guard exceeded';terminate(job['proc'])
    if time.monotonic()-job['start']>args.wall_limit:job['termination_reason']='wall limit exceeded';terminate(job['proc'])
   peak_aggregate=max(peak_aggregate,aggregate)
   if aggregate>AGGREGATE_RSS:
    for job in active.values():job['termination_reason']='aggregate process-tree RSS exceeded3GiB';terminate(job['proc'])
   for name,job in list(active.items()):
    proc=job['proc']
    if proc.poll() is None:continue
    job['log'].close();out=job['out'];check=completion(out,job['cfg']);complete=proc.returncode==0 and job['termination_reason'] is None and check['execution_complete']
    status=dict(status='completed' if complete else 'failed',execution_complete=complete,native_returncode=proc.returncode,termination_reason=job['termination_reason'],started_utc=job['started_utc'],finished_utc=utc(),**{k:v for k,v in check.items() if k!='execution_complete'})
    atomic(out/'candidate_status.json',status)
    if not complete and (out/'statistics.json').exists():shutil.move(str(out/'statistics.json'),str(out/'incomplete_statistics.json'))
    usage=dict(case=name,backend='Athena++ native double HLLD CT CPU',wall_seconds_including_inputs=time.monotonic()-job['start'],exit_code=proc.returncode if proc.returncode else (0 if complete else 86),native_returncode=proc.returncode,process_tree_RSS_cap_MiB=args.rss_mib,sampled_peak_process_tree_RSS_MiB=job['peak_rss']/2**20,address_space_cap_MiB=args.as_mib,CPU_time_cap_s=args.cpu_limit,wall_limit_s=args.wall_limit,core_dump_cap_bytes=0,threads=job['cfg']['threads'],radial_meshblocks=job['cfg'].get('ct_radial_blocks',1),runtime_env={**pin['runtime_env'],'OMP_NUM_THREADS':str(job['cfg']['threads']),'OMP_DYNAMIC':'FALSE'},command=job['command'],nice=15,concurrent_case_limit=args.parallel,termination_reason=job['termination_reason'],resource_usage_source='/usr/bin/time -v resource_usage.txt is authoritative per-process CPU/RSS; no RUSAGE_CHILDREN accumulation',scientific_gates_passed=check.get('scientific_gates_passed',False))
    usage.update(time_usage(out/'resource_usage.txt'))
    atomic(out/'runner_usage.json',usage);finished.append(dict(case=name,execution_complete=complete,scientific_gates_passed=check.get('scientific_gates_passed',False)));del active[name]
   atomic(progress,dict(updated_utc=utc(),phase='Athena CT bounded batch',active=list(active),pending=[x['name'] for x in pending],finished=finished,parallel=args.parallel,aggregate_RSS_cap_bytes=AGGREGATE_RSS,peak_aggregate_RSS_bytes=peak_aggregate))
   if active:time.sleep(.2)
 except BaseException:
  for job in active.values():
   terminate(job['proc']);job['log'].close()
   atomic(job['out']/'candidate_status.json',dict(status='interrupted',execution_complete=False,termination_reason='controller interrupted; child process group terminated',finished_utc=utc()))
  raise
 return finished
