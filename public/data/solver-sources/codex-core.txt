void ecliptic_write_if_due(double day){

 // Multiple requested epochs crossed by one accepted step share one actual frame.
 do{ecliptic_max_lag=std::max(ecliptic_max_lag,std::max(0.,day-ecliptic_next_day));
  ecliptic_schedule<<std::setprecision(17)<<ecliptic_next_day<<','<<day<<','<<name.str()<<'\n';
  ecliptic_next_day+=ecliptic_cadence;
 }while(ecliptic_next_day<=day+1e-12);
 ecliptic_schedule.flush();if(!ecliptic_schedule)throw std::runtime_error("Ecliptic schedule write failed");++ecliptic_frames;
}

void X3ICTBeginStage(MeshBlock*p,int stage,Real rhs_time,Real full_dt){

 if(mode!="uniform"){const auto&goal=stage==1?surface->br0:surface->br1;for(int k=p->ks;k<=p->ke;++k)for(int j=p->js;j<=p->je;++j)maximum_stage_target_error=std::max(maximum_stage_target_error,std::abs(p->pfield->b.x1f(k,j,p->is)-goal[surface->ix(fj(j,p),k-p->ks)]));}

 ++stages;stage_trace<<std::setprecision(17)<<tn<<','<<full_dt<<','<<stage<<','<<rhs_time<<','<<surface->maximum_relative_residual<<'\n';
}
void X3ICTFluxAudit(MeshBlock*p,int stage,Real full_dt){
 std::lock_guard<std::mutex> lock(observer_mutex);++native_rhs_calls;
 auto&f=p->phydro->flux;auto*c=p->pcoord;long double rm=0,re=0;double weight=.5*full_dt;
 for(int k=p->ks;k<=p->ke;++k)for(int j=p->js;j<=p->je;++j)for(int i=p->is;i<=p->ie;++i){
  for(int d=0;d<3;++d){int ih=i+(d==0),jh=j+(d==1),kh=k+(d==2);double al=d==0?c->GetFace1Area(k,j,i):d==1?c->GetFace2Area(k,j,i):c->GetFace3Area(k,j,i);double ah=d==0?c->GetFace1Area(kh,jh,ih):d==1?c->GetFace2Area(kh,jh,ih):c->GetFace3Area(kh,jh,ih);rm-=ah*f[d](IDN,kh,jh,ih)-al*f[d](IDN,k,j,i);re-=ah*f[d](IEN,kh,jh,ih)-al*f[d](IEN,k,j,i);}
  if(GM!=0)re-=.5*((1/(c->x1v(i)*c->x1v(i)))*f[X1DIR](IDN,k,j,i)*GM+(1/(c->x1v(i)*c->x1v(i)))*f[X1DIR](IDN,k,j,i+1)*GM)*c->GetCellVolume(k,j,i);
 }
 integrated_mass+=weight*rm;integrated_energy+=weight*re;
 EndFlux left,right;
 for(int k=p->ks;k<=p->ke;++k)for(int j=p->js;j<=p->je;++j){double al=c->GetFace1Area(k,j,p->is),ar=c->GetFace1Area(k,j,p->ie+1);left.mass+=al*f[X1DIR](IDN,k,j,p->is);left.energy+=al*f[X1DIR](IEN,k,j,p->is);right.mass+=ar*f[X1DIR](IDN,k,j,p->ie+1);right.energy+=ar*f[X1DIR](IEN,k,j,p->ie+1);}
 left_flux[block_index(p)][stage-1]=left;right_flux[block_index(p)][stage-1]=right;
 if(!physical_inner(p))return;

 for(int k=p->ks;k<=p->ke;++k)for(int j=p->js;j<=p->je;++j){int a=fj(j,p),l=k-p->ks;double eth=.5*(surface->et[surface->ix(a,l)]+surface->et[surface->ix(a,l+1)]),eph=.5*(surface->ep[surface->ix(a,l)]+surface->ep[surface->ix(a+1,l)]);auto z=hydro(p->pmy_mesh->time+(stage==2?full_dt:0),rmin,c->x2v(j),c->x3v(k));if(z[1]<=0)continue;double bt=-eph/z[1],bp=eth/z[1],br=p->pfield->b.x1f(k,j,p->is);double sound=gamma_gas*z[4]/z[0],va=(br*br+bt*bt+bp*bp)/z[0],var=br*br/z[0],cf=std::sqrt(.5*(sound+va+std::sqrt(std::max(0.,(sound+va)*(sound+va)-4*sound*var))));min_mach=std::min(min_mach,z[1]/cf);double s=eth*bp-eph*bt,hyd=z[1]*(gamma_gas/(gamma_gas-1)*z[4]+.5*z[0]*z[1]*z[1]),num=f[X1DIR](IEN,k,j,p->is)-hyd,area=c->GetFace1Area(k,j,p->is);ct_work+=weight*area*s;energy_minus_hydro_work+=weight*area*num;max_poynting_relative=std::max(max_poynting_relative,std::abs(num-s)/(std::abs(s)+std::abs(hyd)+1e-99));double bt_actual=p->pfield->bcc(IB2,k,j,p->is-1),bp_actual=p->pfield->bcc(IB3,k,j,p->is-1);double va_actual=(br*br+bt_actual*bt_actual+bp_actual*bp_actual)/z[0],cf_actual=std::sqrt(.5*(sound+va_actual+std::sqrt(std::max(0.,(sound+va_actual)*(sound+va_actual)-4*sound*var))));minimum_native_ghost_mach=std::min(minimum_native_ghost_mach,z[1]/cf_actual);double work_actual=eth*bp_actual-eph*bt_actual,physical_actual=z[1]*(bt_actual*bt_actual+bp_actual*bp_actual);ct_actual_trace_work+=weight*area*work_actual;maximum_actual_trace_work_discrepancy=std::max(maximum_actual_trace_work_discrepancy,std::abs(num-work_actual)/(std::abs(physical_actual)+std::abs(hyd)+1e-99));maximum_hlld_physical_energy_residual=std::max(maximum_hlld_physical_energy_residual,std::abs(num-physical_actual)/(std::abs(physical_actual)+std::abs(hyd)+1e-99));}
}

void MeshBlock::ProblemGenerator(ParameterInput*){
 if(pmy_mesh->adaptive||pmy_mesh->multilevel||loc.level!=pmy_mesh->root_level||loc.lx1<0||loc.lx1>=nradialblocks||je-js+1!=nt||ke-ks+1!=np||ie-is+1!=nr/nradialblocks||loc.lx2!=0||loc.lx3!=0)throw std::runtime_error("Unsupported MeshBlock decomposition");
 for(int i=is;i<=ie;++i){int g=global_i(this,i);radii[g]=pcoord->x1v(i);radial_owner[g]=this;radial_local_i[g]=i;}
 if(physical_inner(this)){for(int j=0;j<nt;++j)thetas[j]=pcoord->x2v(js+j);for(int k=0;k<np;++k)phis[k]=pcoord->x3v(ks+k);}
 const auto&goal=initial_br;auto&b=pfield->b;
 for(int k=ks-NGHOST;k<=ke+NGHOST;++k)for(int j=js-NGHOST;j<=je+NGHOST;++j)for(int i=is-NGHOST;i<=ie+NGHOST+1;++i){double th=pcoord->x2v(j),ph=pcoord->x3v(k),v;if(mode=="uniform"){double lo=pcoord->x2f(j),hi=pcoord->x2f(j+1);v=cfg->d("br_1au_nt",3)*1e-9/BUNIT*.5*(std::pow(std::sin(hi),2)-std::pow(std::sin(lo),2))/(std::cos(lo)-std::cos(hi));}else v=goal[surface->ix(sfj(th),sfk(ph))]*rmin*rmin/(pcoord->x1f(i)*pcoord->x1f(i));b.x1f(k,j,i)=v;}
 for(int k=ks-NGHOST;k<=ke+NGHOST;++k)for(int j=js-NGHOST;j<=je+NGHOST+1;++j)for(int i=is-NGHOST;i<=ie+NGHOST;++i)b.x2f(k,j,i)=mode=="uniform"?-cfg->d("br_1au_nt",3)*1e-9/BUNIT*std::sin(pcoord->x2f(j)):0;
 for(int k=ks-NGHOST;k<=ke+NGHOST+1;++k)for(int j=js-NGHOST;j<=je+NGHOST;++j)for(int i=is-NGHOST;i<=ie+NGHOST;++i)b.x3f(k,j,i)=0;
 pfield->CalculateCellCenteredField(b,pfield->bcc,pcoord,is-NGHOST,ie+NGHOST,js-NGHOST,je+NGHOST,ks-NGHOST,ke+NGHOST);
 for(int k=ks-NGHOST;k<=ke+NGHOST;++k)for(int j=js-NGHOST;j<=je+NGHOST;++j)for(int i=is-NGHOST;i<=ie+NGHOST;++i){auto z=hydro(pmy_mesh->time,pcoord->x1v(i),pcoord->x2v(j),pcoord->x3v(k));phydro->u(IDN,k,j,i)=z[0];phydro->u(IM1,k,j,i)=z[0]*z[1];phydro->u(IM2,k,j,i)=z[0]*z[2];phydro->u(IM3,k,j,i)=z[0]*z[3];double bs=0;for(int d=0;d<3;++d)bs+=std::pow(pfield->bcc(IB1+d,k,j,i),2);phydro->u(IEN,k,j,i)=z[4]/(gamma_gas-1)+.5*z[0]*(z[1]*z[1]+z[2]*z[2]+z[3]*z[3])+.5*bs;}
}
